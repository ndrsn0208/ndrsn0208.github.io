% A shared title/date row. The title may wrap; dates stay aligned at the right.
\newsavebox{\CVTitleBox}
\newif\ifCVHeadingWrapped
\newcommand{\CVHeading}[2]{%
  \sbox{\CVTitleBox}{\begin{minipage}[t]{0.74\linewidth}
    \raggedright\strut #1\strut\par
  \end{minipage}}%
  \ifdim\dp\CVTitleBox>\baselineskip
    \CVHeadingWrappedtrue
  \else
    \CVHeadingWrappedfalse
  \fi
  \par\noindent\usebox{\CVTitleBox}\hfill
  \begin{minipage}[t]{0.24\linewidth}
    \raggedleft\itshape\strut #2\strut\par
  \end{minipage}\par\nobreak\nointerlineskip
}

% Institution or teaching role, date range, and degree/lab/institution.
\newcommand{\CVEntry}[3]{%
  \Needspace{3\baselineskip}%
  \CVHeading{\textbf{#1}}{#2}
  \ifCVHeadingWrapped\vspace{-4.175pt}\else\vspace{-3.3pt}\fi
  {\itshape\strut #3\par}
}
\newcommand{\CVDetail}[1]{%
  \noindent\hspace*{0.235in}#1\par
}

% Paper title, date or venue label, URL (empty if absent in source), authors.
\newcommand{\CVPaperHeading}[4]{%
  \Needspace{3\baselineskip}%
  \CVHeading{%
    \if\relax\detokenize{#3}\relax
      \textbf{#1}%
    \else
      \href{#3}{\textbf{#1}}%
    \fi
  }{#2}
  \ifCVHeadingWrapped\vspace{-4pt}\else\vspace{-3.3pt}\fi
  \strut #4\par
}

% Full-list entry: title, date, URL, authors, publication venue.
\newcommand{\CVPublication}[5]{%
  \begin{samepage}
  \CVPaperHeading{#1}{#2}{#3}{#4}
  \vspace{-1.1pt}
  {\itshape #5\par}
  \end{samepage}
  \CVGap
}

\newcommand{\CVTalk}[4]{%

    \vspace{-1.1pt}
    {\itshape\strut #4\par}
  \fi

}
