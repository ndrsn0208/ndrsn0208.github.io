  {Trust Region Continual Learning as an Implicit Meta-Learner}
  {NeurIPS 2026}
  {https://ndrsn0208.github.io/papers/trust-region-continual-learning.pdf}
  {\CVNameInAuthors\CVEqual, Anant Gupta\CVEqual, Christopher J. MacLellan}
\begin{CVHighlights}
  \item Proved that replay with a Fisher-metric trust region yields a one-step

  \item Improved retention and re-convergence over EWC, replay, FTML, and VR-MCL
    on ImageNet-500 diffusion generation and Continual-World-10 diffusion policies.
\end{CVHighlights}

\CVPaperHeading
  {LeAH: Generalizing Test-Time Agent Harness Adaptation to Unseen Environments}
  {Preprint}
  {https://ndrsn0208.github.io/papers/learning-to-adapt-harnesses.pdf}
  {\CVNameInAuthors, Yingqiang Ge, Hai Wang, Yuhui Xu, Joshua Frandsen, Eric Yang,
   Shancong Fu, Christopher J. MacLellan, Chandan K. Reddy}
\begin{CVHighlights}
  \item Formulated harness adaptation that generalizes to new benchmarks as a joint
    objective and meta-trained a 9B proposer that reads a few failures of an unseen
    benchmark and writes one harness edit, \textbf{without search or weight updates}.
  \item Raised the average score on six unseen benchmarks from 51.4 under the initial
    harness to 57.9 with a frozen agent. Work done during an internship at Amazon.
\end{CVHighlights}
\end{samepage}

\CVPaperHeading
  {Self-Consolidating Language Models}
  {Preprint}
  {https://ndrsn0208.github.io/papers/self-consolidating-language-models.pdf}
  {\CVNameInAuthors\CVEqual, Anant Gupta\CVEqual, Zihan Dong, Christopher J. MacLellan}
\begin{CVHighlights}
  \item Developed a meta-RL post-training framework in which an LLM learns which
    layers to update as new context arrives, balancing knowledge acquisition
    against forgetting.
  \item Improved both acquisition and retention on SQuAD over every evaluated
    baseline; meta-trained only on short LongBench v2 contexts, it generalized to
    longer contexts and outperformed in-context methods there.
\end{CVHighlights}

\begin{samepage}
\CVPaperHeading
  {Test-Time Compositional Generalization in Diffusion Models via Prototype Discovery}
  {Preprint}
  {https://ndrsn0208.github.io/papers/test-time-compositional-generalization.pdf}
  {\CVNameInAuthors, Anant Gupta, Tianyi Zhu, Christopher J. MacLellan}
\begin{CVHighlights}
  \item Developed test-time prototype discovery in pretrained diffusion models: score
    ascent at several noise levels recovers prototypes \textbf{without labels, a
    concept library, or additional training}.
  \item Composed the prototypes that explain a single out-of-distribution query into
    a product of experts; on held-out ColorMNIST and CelebA compositions, samples
    generalized to the unseen class, where sampling from trained concept tokens
    landed on neighboring classes.
\end{CVHighlights}
\end{samepage}

\begin{samepage}
\CVPaperHeading
  {Avoid Catastrophic Forgetting with Rank-1 Fisher from Diffusion Models}
  {ICLR 2026}
  {https://arxiv.org/abs/2509.23593}
  {\CVNameInAuthors\CVEqual, Anant Gupta\CVEqual, Zihan Dong, Christopher J. MacLellan}
\begin{CVHighlights}
  \item Derived and empirically validated a low-SNR rank-1 Fisher approximation
    near convergence, enabling structured EWC with generative replay.
  \item Reduced mean FID-based forgetting by \textbf{67\% on ImageNet-1k}
    (32$\times$32, 20 tasks) versus generative distillation alone; nearly
    eliminated forgetting on MNIST and FashionMNIST.
\end{CVHighlights}
\end{samepage}
