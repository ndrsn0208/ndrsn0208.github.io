  {Self-Consolidating Language Models: Continual Knowledge Incorporation from Context}
  {Preprint}
  {https://arxiv.org/abs/2605.07076}
  {\CVNameInAuthors\CVEqual, Anant Gupta\CVEqual, Zihan Dong, Christopher J. MacLellan}
\begin{CVHighlights}
  \item Developed a meta-RL post-training framework in which an LLM learns which

  \item Improved SQuAD acquisition and retention over sequential SEAL, and

    transferred to longer context streams.

\begin{samepage}
\CVPaperHeading
  {Avoid Catastrophic Forgetting with Rank-1 Fisher from Diffusion Models}
  {ICLR 2026}
  {https://arxiv.org/abs/2509.23593}
  {\CVNameInAuthors\CVEqual, Anant Gupta\CVEqual, Zihan Dong, Christopher J. MacLellan}
\begin{CVHighlights}
  \item Derived and empirically validated a low-SNR rank-1 Fisher approximation
    near convergence, enabling structured EWC with generative replay.
  \item Reduced mean FID-based forgetting by \textbf{67\% on ImageNet-1k}
    (32$\times$32, 20 tasks) versus generative distillation alone; nearly
    eliminated forgetting on MNIST and FashionMNIST.
\end{CVHighlights}
\end{samepage}

  {Test-Time Compositional Generalization in Diffusion Models via Concept Discovery}
  {Preprint}
  {https://arxiv.org/abs/2605.07078}
  {\CVNameInAuthors, Anant Gupta, Tianyi Zhu, Christopher J. MacLellan}
\begin{CVHighlights}
  \item Developed test-time concept discovery and composition from a pretrained
    diffusion model's scores, enabling novel combinations \textbf{without
    retraining the base model}.
  \item Outperformed query-only and nearest-class baselines on ColorMNIST and
    CelebA composition using a product-of-experts sampler and a distilled variant.
\end{CVHighlights}

  {Trust Region Continual Learning as an Implicit Meta-Learner}
  {Preprint}
  {https://arxiv.org/abs/2602.02417}
  {\CVNameInAuthors\CVEqual, Anant Gupta\CVEqual, Christopher J. MacLellan}
\begin{CVHighlights}
  \item Proved that replay with a Fisher-metric trust region yields a one-step
    MAML-like update under local approximations, connecting continual learning
    to meta-learning.
  \item Improved retention and re-convergence over EWC, replay, FTML, and VR-MCL
    on ImageNet-500 diffusion generation and Continual-World-10 diffusion policies.
\end{CVHighlights}
